    isSolved(saved) {
      return Boolean(saved && !this.isRevealed(saved) && (

        saved.isSolved ||
        saved.status === "solved" ||
        saved.status === "complete" ||
        saved.completedAt ||
        saved.solvedAt
      ));
    },

      return Boolean(saved && (saved.revealed || saved.revealedAt));
    },

    isFinished(saved) {
      return Boolean(saved && (
        saved.solved ||
        saved.revealed ||
        saved.completed ||
        saved.isSolved ||
        saved.status === "solved" ||
        saved.status === "complete" ||
        saved.completedAt ||

    },

      return Boolean(
        saved &&
        saved.mappings &&
        Object.keys(saved.mappings).length > 0 &&
        !this.isFinished(saved)
      );
    },

      if (!saved) return null;
      return saved.completedAt || saved.solvedAt || saved.revealedAt || saved.finishedAt || saved.updatedAt || saved.lastPlayedAt || null;

  function loadExternalScript(src, globalName, errorMessage) {
    return new Promise((resolve, reject) => {
      if (globalName && window[globalName]) return resolve();

      const existing = Array.from(document.scripts || []).find(item => item.src === src || item.getAttribute("src") === src);
      if (existing) {
        existing.addEventListener("load", resolve, { once: true });
        existing.addEventListener("error", () => reject(new Error(errorMessage || `Could not load script: ${src}`)), { once: true });
        return;
      }

      const script = document.createElement("script");
      script.src = src;
      script.onload = resolve;
      script.onerror = () => reject(new Error(errorMessage || `Could not load script: ${src}`));
      document.head.appendChild(script);
    });
  }

  async function ensureCore() {
    if (!Core) {
      await loadExternalScript(CORE_URL, "HarePuzzleCore", "Could not load Hare Puzzle Core.");
      Core = window.HarePuzzleCore;
    }

    if (!Core) {
      throw new Error("Hare Puzzle Core is required but was not available.");
    }

    return Core;
  }

  function getNextCryptogramReleaseDate() {
    const now = new Date();
    const future = [...(availablePuzzles || [])]
      .map(item => {
        const parts = String(item.puzzleDate || "").split("-").map(Number);
        if (parts.length !== 3 || parts.some(Number.isNaN)) return null;
        return new Date(parts[0], parts[1] - 1, parts[2], 6, 0, 0, 0);
      })
      .filter(date => date && date > now)
      .sort((a, b) => a - b);

    if (future.length) return future[0];

    const next = new Date();
    next.setHours(6, 0, 0, 0);

    const day = next.getDay();
    const daysUntilSunday = (7 - day) % 7;

    next.setDate(next.getDate() + daysUntilSunday);

    if (now >= next) {
      next.setDate(next.getDate() + 7);
    }

    return next;
  }

  function getCountdownText() {
    const now = new Date();
    const next = getNextCryptogramReleaseDate();
    const diff = Math.max(0, next - now);

    const days = Math.floor(diff / (1000 * 60 * 60 * 24));
    const hours = Math.floor((diff / (1000 * 60 * 60)) % 24);
    const minutes = Math.floor((diff / (1000 * 60)) % 60);
    const seconds = Math.floor((diff / 1000) % 60);

    return (
      String(days).padStart(2, "0") + "d " +
      String(hours).padStart(2, "0") + ":" +
      String(minutes).padStart(2, "0") + ":" +
      String(seconds).padStart(2, "0")
    );
  }

    const status = document.getElementById("hp-message-status");
    const message = textarea.value.trim();

      status.textContent = "Please enter a short message first.";

      submissionType: currentMessageType === "bug" ? "BUG" : "FEEDBACK",

      message

      await fetch(GOOGLE_APPS_SCRIPT_URL, {

        mode: "no-cors",

  }

  async function loadPuzzleFromMeta(selectedMeta, options = {}) {
    const { updateUrl = false, scroll = false, replaceUrl = false } = options;

    if (!selectedMeta) throw new Error("Puzzle not found in index.json.");

    mount.innerHTML = `<p style="text-align:center; padding:20px; font-weight:800;">Loading Cryptogram...</p>`;

    const puzzleData = await Core.getPuzzleDataForMeta(selectedMeta, cryptogramCoreConfig);

    puzzleData.puzzleTitle = `Cryptogram Puzzle #${puzzleData.puzzleId}`;
    puzzleData.morePuzzlesUrl = puzzleData.morePuzzlesUrl || "https://www.harepublishing.com/online-puzzles";
    puzzleData.shopUrl = puzzleData.shopUrl || "https://www.harepublishing.com/shop";

    currentPuzzleData = puzzleData;
    window.HareCryptogramData = puzzleData;

    titleEl.textContent = puzzleData.puzzleTitle;
    dateEl.textContent = Core.formatDate(puzzleData.puzzleDate || selectedMeta.puzzleDate);

    await Core.loadScript(ENGINE_URL, {
      errorMessage: "Could not load Cryptogram engine."
    });

    if (!window.HareCryptogramEngine || typeof window.HareCryptogramEngine.init !== "function") {
      throw new Error("Cryptogram engine loaded but did not initialize correctly.");
    }

    delete container.dataset.hpCryptogramMounted;

    window.HareCryptogramEngine.init({
      containerId: "hp-cryptogram-container",
      dataObject: puzzleData
    });

    renderPlayerStats();
    renderNextCryptogramButton();

    if (updateUrl) {
      Core.updatePuzzleUrl(puzzleData.puzzleId, {
        mode: replaceUrl ? "replace" : "push",
        state: { hpCryptogramPuzzleId: String(puzzleData.puzzleId) }
      });
    }

    if (scroll) {
      document.getElementById("hp-cryptogram-platform")?.scrollIntoView({ behavior: "smooth", block: "start" });
    }
  }

  async function loadPuzzleById(puzzleId, options = {}) {
    const selectedMeta = availablePuzzles.find(item => String(item.puzzleId) === String(puzzleId));

    if (!selectedMeta) {
      showError("This puzzle is not available yet.");
      return;
    }

    try {
      await loadPuzzleFromMeta(selectedMeta, {
        updateUrl: options.updateUrl !== false,
        replaceUrl: Boolean(options.replaceUrl),
        scroll: Boolean(options.scroll)
      });
    } catch (error) {
      console.error(error);
      showError(error.message);
    }
  }

  async function loadInitialPuzzle() {
    const index = await Core.loadIndex(cryptogramCoreConfig);
    const requestedPuzzleId = Core.getPuzzleIdFromUrl();

    availablePuzzles = Core.getAvailablePuzzles(index, { sort: "ascending" });
    window.HareCryptogramIndex = index;

    let selectedMeta;

    if (requestedPuzzleId) {
      selectedMeta = availablePuzzles.find(item => String(item.puzzleId) === String(requestedPuzzleId));
    } else {
      selectedMeta = Core.getNewestAvailablePuzzle(index);
    }

    if (!selectedMeta) {
      if (requestedPuzzleId) {
        throw new Error("This puzzle is not available yet.");
      }
      throw new Error("Puzzle not found in index.json.");
    }

    await loadPuzzleFromMeta(selectedMeta, { updateUrl: false, scroll: false });
  }

  window.HareCryptogramLoadPuzzle = loadPuzzleById;

  document.getElementById("hp-side-help").addEventListener("click", () => {
    if (window.HareCryptogramEngine?.openHelp) {
      window.HareCryptogramEngine.openHelp("hp-cryptogram-container");
    }
  });

  document.querySelectorAll("[data-hp-message]").forEach(btn => {
    btn.addEventListener("click", () => openMessageModal(btn.dataset.hpMessage));
  });

  document.getElementById("hp-message-close").addEventListener("click", closeMessageModal);
  document.getElementById("hp-message-send").addEventListener("click", sendMessage);

  document.getElementById("hp-message-modal").addEventListener("click", e => {
    if (e.target.id === "hp-message-modal") closeMessageModal();
  });

  if (nextCryptogramSlot) {
    nextCryptogramSlot.addEventListener("click", e => {
      const btn = e.target.closest("[data-hp-load-puzzle]");
      if (!btn) return;
      const puzzleId = btn.getAttribute("data-hp-load-puzzle");
      if (puzzleId) loadPuzzleById(puzzleId, { scroll: false });
    });
  }

  window.addEventListener("popstate", () => {
    const puzzleId = Core?.getPuzzleIdFromUrl();
    if (puzzleId) {
      loadPuzzleById(puzzleId, { updateUrl: false, scroll: false });
      return;
    }

    const newest = availablePuzzles[availablePuzzles.length - 1];
    if (newest) loadPuzzleFromMeta(newest, { updateUrl: false, scroll: false });
  });

  window.addEventListener("hare:cryptogram-state-change", () => {
    renderPlayerStats();
    renderNextCryptogramButton();
  });

  try {
    await ensureCore();
    await loadInitialPuzzle();

    if (timerIntervalId) clearInterval(timerIntervalId);
    timerIntervalId = setInterval(() => {
      const timer = document.getElementById("hp-cryptogram-countdown-time");
      if (timer) timer.textContent = getCountdownText();
    }, 1000);

    const timer = document.getElementById("hp-cryptogram-countdown-time");
    if (timer) timer.textContent = getCountdownText();

  } catch (error) {
    console.error(error);
    showError(error.message);
  }
})();
</script>
